# ── Models ──────────────────────────────────────────────────────────────────
class SignupRequest(BaseModel):
    name: str = ""
    email: str
    password: str
    date_of_birth: date
    weight_kg: float = Field(gt=0, le=500)
    allergies: list[str] = Field(default_factory=list, max_length=30)

    @field_validator("date_of_birth")
    @classmethod
    def validate_date_of_birth(cls, value: date) -> date:
        if value > date.today():
            raise ValueError("Date of birth cannot be in the future")
        return value

    @field_validator("name")
    @classmethod
    def validate_name(cls, value: str) -> str:
        if not value.strip():
            raise ValueError("Name cannot be empty")
        return value

    @field_validator("password")
    @classmethod
    def validate_password(cls, value: str) -> str:
        if len(value) < 8:
            raise ValueError("Password must be at least 8 characters")
        return value

def normalize_allergies(allergies: list[str]) -> list[str]:
    normalized = []
    seen = set()
    for allergy in allergies:
        term = allergy.strip()
        key = term.casefold()
        if term and key not in {"none", "n/a", "na"} and key not in seen:
            normalized.append(term)
            seen.add(key)
    return normalized


async def authenticated_user(
    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
) -> dict:
    if credentials is None:
        raise HTTPException(status_code=401, detail="Authentication required")
    token_hash = hashlib.sha256(credentials.credentials.encode("utf-8")).hexdigest()
    user = await users_collection.find_one({"auth_token_hash": token_hash})
    if user is None:
        raise HTTPException(status_code=401, detail="Invalid or expired session")
    return user

# ── SSE helper ──────────────────────────────────────────────────────────────
def sse_event(event: str, data: dict | str) -> str:
    """Format a Server-Sent Event."""
    payload = json.dumps(data) if isinstance(data, dict) else json.dumps({"message": data})
    return f"event: {event}\ndata: {payload}\n\n"


# ── Optimize endpoint (SSE stream) ─────────────────────────────────────────
@app.post("/api/optimize")
async def optimize(
    request: OptimizeRequest,
    user: dict = Depends(authenticated_user),
):
    """Stream the optimization pipeline as SSE events."""

    async def event_stream():
        from graph.graph import build_graph

        graph = build_graph()
        config = {"configurable": {"thread_id": str(uuid.uuid4())}}

        user_input = {
            "menu": [{"name": item.name, "price": item.price} for item in request.menu],
            "budget": request.budget,
            "protein_target": request.protein_target,
            "dietary_preference": request.dietary_preference,
            "allergies": normalize_allergies(user.get("allergies") or []),
            "protein_estimates": {},
            "valid_combinations": [],
            "best_combination": None,
            "recommendation": "",
        }

        language = request.language

        async def graph_events():
            try:
                async for graph_event in graph.astream(user_input, config=config):
                    yield graph_event
            except ValueError as exc:
                logger.info("Optimization request could not be completed: %s", exc)
                yield {"__error__": str(exc)}
            except Exception:
                logger.exception("Optimization graph failed")
                yield {"__error__": "Optimization failed. Check the backend logs for details."}

        # Stream status messages without adding artificial latency.
        yield sse_event("status", {"step": "typing", "message": "🔍 Typing..."})

        yield sse_event("status", {"step": "validating", "message": "✅ Validating your input..."})

        async for event in graph_events():
            if event.get("__error__"):
                yield sse_event(
                    "error",
                    event["__error__"],
                )
                return

            for node_name, node_output in event.items():
                if node_name == "parse_input":
                    yield sse_event("status", {"step": "analyzing", "message": "🧬 Analyzing protein content..."})

                elif node_name == "estimate_protein":
                    estimates = node_output.get("protein_estimates", {})
                    yield sse_event("protein_estimates", {
                        "estimates": {k: v for k, v in estimates.items()},
                    })
                    yield sse_event("status", {"step": "optimizing", "message": "📊 Finding best combinations..."})

                elif node_name == "optimize":
                    combos = node_output.get("valid_combinations", [])
                    best = node_output.get("best_combination")
                    yield sse_event("combinations", {
                        "combinations": combos,
                        "best": best,
                    })
                    yield sse_event("status", {"step": "preparing", "message": "🤖 Preparing AI recommendation..."})

                elif node_name == "recommend":
                    recommendation = node_output.get("recommendation", "")

                    # If language is not English, translate the recommendation
                    if language.lower() != "english":
                        yield sse_event("status", {"step": "translating", "message": f"🌐 Translating to {language}..."})
                        translate_llm = ChatGoogleGenerativeAI(
                            model="gemini-2.5-flash", temperature=0.3
                        )
                        translated = translate_llm.invoke(
                            f"Translate the following nutrition recommendation to {language}. "
                            f"Keep all numbers, food names, and ₹ symbols intact. "
                            f"Make it sound natural in {language}:\n\n{recommendation}"
                        )
                        recommendation = translated.content

                    yield sse_event("recommendation", {"text": recommendation})

        yield sse_event("done", {"message": "✨ Analysis complete!"})

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )


# ── Chat endpoint ──────────────────────────────────────────────────────────
@app.post("/api/chat")
async def chat_endpoint(
    request: ChatRequest,
    user: dict = Depends(authenticated_user),
):
    """Handle conversational follow-ups via LLM."""
    sys_prompt = "You are NutriX, a helpful nutrition AI assistant."
    allergies = normalize_allergies(user.get("allergies") or [])
    if request.context:
        sys_prompt += f"\n\nCurrent Session Context: Budget: ₹{request.context.get('budget', '?')}, Protein Target: {request.context.get('protein_target', '?')}g. Menu provided: {request.context.get('menu', '?')}."

    if allergies:
        sys_prompt += (
            "\n\nFood allergy safety constraint: never recommend or include food items "
            "containing these user-reported allergens: "
            + ", ".join(allergies)
            + ". If ingredients are unclear, advise the user to confirm with the restaurant."
        )

    langchain_msgs = [SystemMessage(content=sys_prompt)]
    
    for msg in request.messages:
        role = msg.get("role", "")
        content = msg.get("content", "")
        if role == "user":
            langchain_msgs.append(HumanMessage(content=content))
        elif role == "assistant":
            langchain_msgs.append(AIMessage(content=content))
            
    response = vision_llm.invoke(langchain_msgs)
    return {"reply": response.content}


# ── Image recognition endpoint ─────────────────────────────────────────────
@app.post("/api/recognize-food")
async def recognize_food(file: UploadFile = File(...)):
    """Extract menu items and prices from an uploaded menu image."""
    allowed_content_types = {"image/jpeg", "image/png", "image/webp"}
    if file.content_type not in allowed_content_types:
        raise HTTPException(
            status_code=415,
            detail="Upload a JPEG, PNG, or WebP menu image.",
        )

    contents = await file.read(10 * 1024 * 1024 + 1)
    if len(contents) > 10 * 1024 * 1024:
        raise HTTPException(status_code=413, detail="Menu images must be 10 MB or smaller.")
    if not contents:
        raise HTTPException(status_code=400, detail="The uploaded menu image is empty.")

    b64_image = base64.b64encode(contents).decode("utf-8")

    message = HumanMessage(
        content=[
            {
                "type": "text",
                "text": (
                    "Read this restaurant food menu image and extract each distinct food item "
                    "with its listed price converted to a number in Indian Rupees. Exclude "
                    "descriptions, add-ons, and non-food items. If an item's price is not "
                    "legible or not listed, estimate a reasonable price in Indian Rupees. "
                    "Return ONLY a JSON array of objects with 'name' and 'price' keys. "
                    "Example: [{\"name\": \"Chicken Rice\", \"price\": 120}, {\"name\": \"Dal Tadka\", \"price\": 80}] "
                    "Return ONLY the JSON array, no markdown, no explanation."
                ),
            },
            {
                "type": "image_url",
                "image_url": {"url": f"data:{file.content_type};base64,{b64_image}"},
            },
        ]
    )

    response = vision_llm.invoke([message])
    text = response.content.strip()

    # Clean markdown code fences if present
    if text.startswith("```"):
        text = text.split("\n", 1)[1] if "\n" in text else text[3:]
        if text.endswith("```"):
            text = text[:-3]
        text = text.strip()

    try:
        parsed_items = json.loads(text)
    except json.JSONDecodeError as exc:
        logger.exception("Could not parse menu image recognition response")
        raise HTTPException(status_code=502, detail="Could not read menu items from this image.") from exc

    if not isinstance(parsed_items, list):
        raise HTTPException(status_code=502, detail="Menu recognition returned an invalid item list.")

    items = []
    for item in parsed_items:
        if not isinstance(item, dict):
            continue
        name = item.get("name")
        price = item.get("price")
        if not isinstance(name, str) or not name.strip():
            continue
        try:
            numeric_price = float(price)
        except (TypeError, ValueError):
            continue
        if not math.isfinite(numeric_price) or numeric_price <= 0:
            continue
        items.append({"name": name.strip(), "price": numeric_price})

    if not items:
        raise HTTPException(status_code=422, detail="No food items with prices could be read from this image.")
    return {"items": items}

client = motor.motor_asyncio.AsyncIOMotorClient(
    os.environ["MONGO_URI"], serverSelectionTimeoutMS=5000
)
db = client.nutriX
users_collection = db.users
conversations_collection = db.conversations


def profile_response(user: dict) -> dict:
    date_of_birth_value = user.get("date_of_birth")
    age = None
    if date_of_birth_value:
        date_of_birth = date.fromisoformat(date_of_birth_value)
        today = date.today()
        age = today.year - date_of_birth.year - (
            (today.month, today.day) < (date_of_birth.month, date_of_birth.day)
        )
    return {
        "name": user.get("name", ""),
        "email": user["email"],
        "dateOfBirth": date_of_birth_value,
        "weightKg": user.get("weight_kg"),
        "age": age,
        "allergies": user.get("allergies", []),
    }


async def issue_auth_token(user_id) -> str:
    token = secrets.token_urlsafe(32)
    token_hash = hashlib.sha256(token.encode("utf-8")).hexdigest()
    await users_collection.update_one(
        {"_id": user_id},
        {"$set": {"auth_token_hash": token_hash}},
    )
    return token

@app.post("/api/auth/signup")
async def register(req: SignupRequest):
    existing = await users_collection.find_one({"email": req.email})
    if existing:
        raise HTTPException(status_code=400, detail="User with this email already exists")
    
    hashed_pwd = bcrypt.hashpw(req.password.encode('utf-8'), bcrypt.gensalt()).decode('utf-8')
    new_user = {
        "name": req.name,
        "email": req.email,
        "password": hashed_pwd,
        "date_of_birth": req.date_of_birth.isoformat(),
        "weight_kg": req.weight_kg,
        "allergies": normalize_allergies(req.allergies),
    }
    result = await users_collection.insert_one(new_user)
    token = await issue_auth_token(result.inserted_id)
    
    return {
        "token": token,
        "user": profile_response(new_user),
    }

@app.post("/api/auth/login")
async def login_route(req: LoginRequest):
    user = await users_collection.find_one({"email": req.email})
    if not user:
        raise HTTPException(status_code=401, detail="Invalid email or password")
    
    if not bcrypt.checkpw(req.password.encode('utf-8'), user["password"].encode('utf-8')):
        raise HTTPException(status_code=401, detail="Invalid email or password")
        
    token = await issue_auth_token(user["_id"])
    return {"token": token, "user": profile_response(user)}


@app.get("/api/auth/profile")
async def get_profile(user: dict = Depends(authenticated_user)):
    return profile_response(user)


@app.put("/api/auth/profile")
async def update_profile(
    req: ProfileUpdateRequest,
    user: dict = Depends(authenticated_user),
):
    updated = {
        "name": req.name.strip(),
        "date_of_birth": req.date_of_birth.isoformat(),
        "weight_kg": req.weight_kg,
        "allergies": normalize_allergies(req.allergies),
    }
    await users_collection.update_one({"_id": user["_id"]}, {"$set": updated})
    return profile_response({**user, **updated})

@app.get("/api/conversations/{conversation_id}")
async def get_conversation(
    conversation_id: str,
    email: str = Query(..., min_length=3),
):
    """Fetch one conversation only when it belongs to the supplied email."""
    from bson import ObjectId

    if not ObjectId.is_valid(conversation_id):
        raise HTTPException(status_code=400, detail="Invalid conversation id")

    conversation = await conversations_collection.find_one(
        {
            "_id": ObjectId(conversation_id),
            "email": email.strip().lower(),
        }
    )
    if conversation is None:
        raise HTTPException(status_code=404, detail="Conversation not found")

    return {
        "id": str(conversation["_id"]),
        "title": conversation.get("title", "New conversation"),
        "createdAt": conversation.get("createdAt"),
        "updatedAt": conversation.get("updatedAt"),
        "messages": conversation.get("messages", []),
    }


def conversation_title(title: str, messages: list[dict]) -> str:
    """Derive a useful title for old conversations with generic saved titles."""
    if title.strip() and title.strip().lower() not in {
        "new conversation",
        "optimization request:",
    }:
        return title.strip()

    first_user_message = next(
        (
            message.get("content", "")
            for message in messages
            if message.get("role") == "user"
        ),
        "",
    )
    menu_match = re.search(
        r"(?:^|\n)Menu items:[ \t]*\r?\n([\s\S]*?)(?:\r?\n[ \t]*\r?\n|$)",
        first_user_message,
        re.IGNORECASE,
    )
    if menu_match:
        items = [
            match.group(1).strip()
            for line in menu_match.group(1).splitlines()
            if (match := re.match(r"^\s*[-•]?\s*(.+?)\s*\(₹?\s*\d+(?:\.\d+)?\)\s*$", line))
        ]
        if items:
            suffix = " + more" if len(items) > 3 else ""
            return f"Menu: {' + '.join(items[:3])}{suffix}"[:60]

    item_match = re.search(r"(?:^|\n)Items:\s*(.+)", first_user_message, re.IGNORECASE)
    if item_match:
        items = [
            re.sub(r"\s*\([^)]*\)", "", item).strip()
            for item in item_match.group(1).split(",")
        ]
        items = [item for item in items if item]
        if items:
            budget_match = re.search(
                r"(?:^|\n)Budget:\s*₹?([\d.]+)",
                first_user_message,
                re.IGNORECASE,
            )
            plan = f"{'₹' + budget_match.group(1) + ' plan: ' if budget_match else 'Meal plan: '}"
            suffix = " + more" if len(items) > 3 else ""
            return f"{plan}{' + '.join(items[:3])}{suffix}"[:60]

    for line in first_user_message.splitlines():
        candidate = line.strip()
        if candidate and not re.match(
            r"^(optimization request|budget:|protein target:|food preference:|items:)",
            candidate,
            re.IGNORECASE,
        ):
            return candidate[:60]
    return "New conversation"


@app.post("/api/conversations")
async def create_conversation(req: ConversationRequest):
    """Create a conversation owned by the supplied email address."""
    now = datetime.now(timezone.utc)
    result = await conversations_collection.insert_one({
        "email": req.email.strip().lower(),
        "title": req.title.strip() or "New conversation",
        "messages": req.messages,
        "createdAt": now,
        "updatedAt": now,
    })
    return {"id": str(result.inserted_id), "createdAt": now, "updatedAt": now}


@app.put("/api/conversations/{conversation_id}")
async def update_conversation(conversation_id: str, req: ConversationRequest):
    """Update messages only when the conversation belongs to the supplied email."""
    from bson import ObjectId

    if not ObjectId.is_valid(conversation_id):
        raise HTTPException(status_code=400, detail="Invalid conversation id")

    now = datetime.now(timezone.utc)
    result = await conversations_collection.update_one(
        {
            "_id": ObjectId(conversation_id),
            "email": req.email.strip().lower(),
        },
        {
            "$set": {
                "title": req.title.strip() or "New conversation",
                "messages": req.messages,
                "updatedAt": now,
            }
        },
    )
    if result.matched_count == 0:
        raise HTTPException(status_code=404, detail="Conversation not found")
    return {"ok": True, "updatedAt": now}


@app.delete("/api/conversations/{conversation_id}")
async def delete_conversation(
    conversation_id: str,
    email: str = Query(..., min_length=3),
):
    """Delete only the signed-in user's selected conversation."""
    from bson import ObjectId

    if not ObjectId.is_valid(conversation_id):
        raise HTTPException(status_code=400, detail="Invalid conversation id")

    result = await conversations_collection.delete_one(
        {
            "_id": ObjectId(conversation_id),
            "email": email.strip().lower(),
        }
    )
    if result.deleted_count == 0:
        raise HTTPException(status_code=404, detail="Conversation not found")
    return {"ok": True}
